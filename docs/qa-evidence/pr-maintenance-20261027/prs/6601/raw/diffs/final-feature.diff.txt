+function classSyntaxCount(code: string): number {
+  let count = 0;
+  const visit = (node: unknown): void => {
+    if (Array.isArray(node)) return node.forEach(visit);
+    if (node === null || typeof node !== "object") return;
+    const { type } = node as { type?: string };
