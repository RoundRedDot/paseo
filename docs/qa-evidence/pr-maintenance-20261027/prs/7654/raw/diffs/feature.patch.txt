+/**
+ * What a plugin screen puts in its header. Each mounted `ScreenHeaderActions` holds an entry;
+ * the last one mounted is shown. `present` changes only when that flips, so the screen can
+ * decide whether to draw the actions island without re-rendering on every content update.
+ */
+export interface ScreenHeaderSlot {
+  set(owner: string, content: ReactNode): void;
+  remove(owner: string): void;
+  subscribe(listener: () => void): () => void;
+  content(): ReactNode;
+  present(): boolean;
+}
+
+export function createScreenHeaderSlot(): ScreenHeaderSlot {
+  const entries = new Map<string, ReactNode>();
+  const listeners = new Set<() => void>();
+  let current: ReactNode = null;
+  const publish = () => {
+    current = entries.size > 0 ? Array.from(entries.values()).at(-1) : null;
+    for (const listener of listeners) listener();
+  };
+  return {
+    set(owner, content) {
+      entries.set(owner, content);
+      publish();
+    },
+    remove(owner) {
+      if (entries.delete(owner)) publish();
+    },
+    subscribe(listener) {
+      listeners.add(listener);
+      return () => listeners.delete(listener);
+    },
+    content: () => current,
+    present: () => entries.size > 0,
+  };
+}

+### Screen header actions
+
+A screen can put its own buttons in its header, beside Paseo's: in the actions island on compact
+layouts, before **Close** on wide ones. Render `ScreenHeaderActions` anywhere in the screen and
+put `ScreenHeaderButton`s in it. They read and set the screen's state like any other child:
+
+```tsx
+import type { PluginScreenProps } from "@getpaseo/plugin/client";
+import { ScreenHeaderActions, ScreenHeaderButton } from "@getpaseo/plugin/client/ui";
+import { useState } from "react";
+
+export function Main(props: PluginScreenProps) {
+  const [searching, setSearching] = useState(false);
+  return (
+    <>
+      <ScreenHeaderActions>
+        <ScreenHeaderButton
+          icon="Search"
+          label="Search"
+          active={searching}
+          onPress={() => setSearching((on) => !on)}
+        />
+      </ScreenHeaderActions>
+      <Results searching={searching} />
+    </>
+  );
+}
+```
+
+`ScreenHeaderActions` draws nothing where it is rendered, and nothing outside a plugin screen.
+Its children update with each render of the screen. With several mounted, the last one shows.
+The header renders them outside the screen's tree: `usePaseo`, `useRpc`, translations, the query
+client and toasts work there, but a React context the plugin provides inside the screen does not
+reach them; pass such values as props.
+
+`ScreenHeaderButton` draws an icon button with the header's own size, highlight and tooltip:
+
+| Prop       | Type         | Required | Behavior                                              |
+| ---------- | ------------ | -------- | ----------------------------------------------------- |
+| `icon`     | `PluginIcon` | Yes      | A Lucide icon name, inline SVG, or a component.       |
+| `label`    | `string`     | Yes      | Accessibility label, and the tooltip on wide layouts. |
+| `onPress`  | `() => void` | Yes      | Runs on press.                                        |
+| `active`   | `boolean`    | No       | Keeps the button highlighted, as for an open search.  |
+| `disabled` | `boolean`    | No       | Dims the button and ignores presses.                  |
+| `testID`   | `string`     | No       | Test identifier.                                      |

+路由、标题栏、关闭操作、主机选择器、错误边界和 query client 由 Paseo 负责。界面主体由插件负责，插件也可以用 [`ScreenHeaderActions`](#screen-header-actions) 往标题栏添加按钮。
 
 `addScreen({ id, title, Component })` 注册一个界面。`title` 是界面标题栏的[显示文本](#runtime-translations)，也可以是接收界面参数并返回显示文本的函数。参数或语言变化时，该函数会再次运行。无效标题和空字符串标题会被拒绝。如果标题函数抛出错误或解析为空字符串，会改为显示界面 ID；界面仍然会渲染。
 
 `openScreen({ screenId, params })` 以参数打开一个界面，例如机器人界面要显示的那个机器人。参数保存在界面 URL 的查询字符串中，因此重新加载、前进后退以及指向该界面的链接都会保留它们。任何字符串键都可以；值必须是字符串，其他类型会抛出错误。用其他参数打开同一个界面会显示新的参数。
 
+### 界面标题栏操作 {#screen-header-actions}
+
+界面可以把自己的按钮放进标题栏，排在 Paseo 自带的按钮旁边：紧凑布局中放在操作区，较宽的布局中放在 **关闭** 之前。在界面中任意位置渲染 `ScreenHeaderActions`，并在其中放入 `ScreenHeaderButton`。这些按钮和其他子组件一样读写界面的状态：
+
+```tsx
+import type { PluginScreenProps } from "@getpaseo/plugin/client";
+import { ScreenHeaderActions, ScreenHeaderButton } from "@getpaseo/plugin/client/ui";
+import { useState } from "react";
+
+export function Main(props: PluginScreenProps) {
+  const [searching, setSearching] = useState(false);
+  return (
+    <>
+      <ScreenHeaderActions>
+        <ScreenHeaderButton
+          icon="Search"
+          label="Search"
+          active={searching}
+          onPress={() => setSearching((on) => !on)}
+        />
+      </ScreenHeaderActions>
+      <Results searching={searching} />
+    </>
+  );
+}
+```
+
+`ScreenHeaderActions` 在其渲染位置不绘制任何内容，在插件界面之外也不绘制任何内容。界面每次渲染时，它的子组件都会随之更新。同时挂载多个时，显示最后挂载的那个。标题栏在界面的组件树之外渲染这些子组件：`usePaseo`、`useRpc`、翻译、query client 和 toast 在其中都可用，但插件在界面内部提供的 React context 无法传到那里；这类值请通过 props 传入。
+
+`ScreenHeaderButton` 绘制一个与标题栏自身按钮尺寸、高亮和提示一致的图标按钮：
+
+| 属性       | 类型         | 必填 | 行为                                   |
+| ---------- | ------------ | ---- | -------------------------------------- |
+| `icon`     | `PluginIcon` | 是   | Lucide 图标名、内联 SVG 或组件。       |
+| `label`    | `string`     | 是   | 无障碍标签，在较宽的布局中也用作提示。 |
+| `onPress`  | `() => void` | 是   | 按下时运行。                           |
+| `active`   | `boolean`    | 否   | 保持按钮高亮，例如搜索已展开时。       |
+| `disabled` | `boolean`    | 否   | 按钮变暗并忽略按下操作。               |
+| `testID`   | `string`     | 否   | 测试标识。                             |
+
+其他子组件按原样渲染；请让它们不超过标题栏的高度。在不提供这些组件的宿主上，导入结果为 `undefined`；如果插件需要支持较旧版本的 Paseo，请在渲染前先检查。
+
 ### 侧边栏项 {#sidebar-items}
 
 `addSidebarHeaderItem` 在侧边栏顶部的列表中添加一行。`addSidebarFooterItem` 在底部添加一行，位于 **添加 project** 和底部图标行之间。图标行是固定的；插件不能向其中添加内容。两者都接受 `{ id, title, Component }`。`title` 是该项在 **设置 → 侧边栏** 中的标签（用户在那里重新排序和隐藏各项），也是 `SidebarRow` 的默认标签和无障碍标签。设置页面为这些项显示一个通用的插件图标；来自已弃用的 `addSidebarItem` 的行保留其注册时的图标。
