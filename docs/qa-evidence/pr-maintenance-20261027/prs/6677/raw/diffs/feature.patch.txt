+import { useCallback } from "react";
+import { RewindMenu, type RewindMode } from "@/components/rewind/rewind-menu";
+import { useRewindAgentMutation } from "@/components/rewind/use-rewind-agent-mutation";
+import { useSessionStore } from "@/stores/session-store";
+import { selectPluginRewindCapabilities } from "./rewind-target";
+
+export type PluginRewindMenuProps = import("@getpaseo/plugin/client/ui").RewindMenuProps;
+
+/**
+ * The host's own rewind menu for a prompt a plugin draws. The menu offers the
+ * modes the agent's provider declares (Claude: conversation, files, both;
+ * Codex, OMP and Pi: conversation; OpenCode: both; none for ACP) and the
+ * rewind runs through the same mutation as the host bubble: the timeline is
+ * refetched and a conversation rewind restores the prompt to the composer.
+ */
+export function PluginRewindMenu({ hostId, agentId, messageId, text }: PluginRewindMenuProps) {
+  const client = useSessionStore((state) => state.sessions[hostId]?.client ?? null);
+  const capabilities = useSessionStore((state) =>
+    selectPluginRewindCapabilities(state.sessions, hostId, agentId),
+  );
+  const mutation = useRewindAgentMutation({ serverId: hostId, agentId, client, messageId });
+  const { rewindAgent } = mutation;
+  const handleRewind = useCallback(
+    (input: { mode: RewindMode; rewoundText: string }) => rewindAgent(input),
+    [rewindAgent],
+  );
+
+  if (!capabilities || !messageId) return null;
+  return (
+    <RewindMenu
+      capabilities={capabilities}
+      rewoundText={text}
+      isPending={mutation.isPending}
+      onRewind={handleRewind}
+    />
+  );
+}

+import { describe, expect, test } from "vitest";
+import type { SessionState } from "@/stores/session-store";
+import { selectPluginRewindCapabilities } from "./rewind-target";
+
+const claude = {
+  supportsRewindConversation: true,
+  supportsRewindFiles: true,
+  supportsRewindBoth: true,
+};
+const codex = {
+  supportsRewindConversation: true,
+  supportsRewindFiles: false,
+  supportsRewindBoth: false,
+};
+
+function session(
+  agents: Record<string, object>,
+  details: Record<string, object> = {},
+): SessionState {
+  const toMap = (entries: Record<string, object>) =>
+    new Map(Object.entries(entries).map(([id, capabilities]) => [id, { id, capabilities }]));
+  return { agents: toMap(agents), agentDetails: toMap(details) } as unknown as SessionState;
+}
+
+describe("selectPluginRewindCapabilities", () => {
+  test("reads the agent's own provider capabilities on the given host", () => {
+    const sessions = { a: session({ claude: claude }), b: session({ codex: codex }) };
+    expect(selectPluginRewindCapabilities(sessions, "a", "claude")).toBe(claude);
+    expect(selectPluginRewindCapabilities(sessions, "b", "codex")).toBe(codex);
+  });
+
+  test("falls back to agent details, as other plugin surfaces do", () => {
+    expect(
+      selectPluginRewindCapabilities({ a: session({}, { archived: codex }) }, "a", "archived"),
+    ).toBe(codex);
+  });
+
+  test("keeps capabilities scoped to the selected host for equal agent ids", () => {
+    const sessions = { a: session({ agent: claude }), b: session({ agent: codex }) };
+    expect(selectPluginRewindCapabilities(sessions, "a", "agent")).toBe(claude);
+    expect(selectPluginRewindCapabilities(sessions, "b", "agent")).toBe(codex);
+  });
+
+  test("uses the live agent capabilities ahead of cached details", () => {
+    const sessions = { a: session({ agent: codex }, { agent: claude }) };
+    expect(selectPluginRewindCapabilities(sessions, "a", "agent")).toBe(codex);
+  });
+
+  test("is null for an unknown host or agent", () => {
+    const sessions = { a: session({ claude: claude }) };
+    expect(selectPluginRewindCapabilities(sessions, "missing", "claude")).toBeNull();
+    expect(selectPluginRewindCapabilities(sessions, "a", "missing")).toBeNull();

+import type { AgentCapabilityFlags } from "@getpaseo/protocol/agent-types";
+import type { SessionState } from "@/stores/session-store";
+
+/** The agent's rewind capabilities on a host, or null while the agent is unknown. */
+export function selectPluginRewindCapabilities(
+  sessions: Record<string, SessionState | undefined>,
+  hostId: string,
+  agentId: string,
+): AgentCapabilityFlags | null {
+  const session = sessions[hostId];
+  const agent = session?.agents.get(agentId) ?? session?.agentDetails.get(agentId);
+  return agent?.capabilities ?? null;
+}

+### Keep rewind on a replaced prompt
+
+A renderer that replaces `user_message` items takes the host bubble's rewind menu with it. Pass the
+item's `messageId` through `data` and draw the host's `RewindMenu` beside the prompt. It offers only
+the modes the agent's provider supports (conversation, files, or both) and renders nothing when
+there are none or the prompt has no `messageId` yet. A rewind runs exactly as from the host bubble:
+the timeline is refetched, and a conversation rewind puts `text` back in an empty composer.
+`RewindMenu` is undefined on hosts that predate it, so check for it before rendering.
+
+```tsx
+import { RewindMenu } from "@getpaseo/plugin/client/ui";
+
+function Prompt({ agentId, host, item }: PluginTimelineItemProps<PromptData>) {
+  return (
+    <View>
+      <Text>{item.data.text}</Text>
+      {RewindMenu ? (
+        <RewindMenu
+          hostId={host.id}
+          agentId={agentId}
+          messageId={item.data.messageId}
+          text={item.data.text}
+        />
+      ) : null}
+    </View>
+  );
+}
+```
