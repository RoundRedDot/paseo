const root=process.cwd(),res=path.join(root,'packages/desktop/release/mac-arm64/Paseo.app/Contents/Resources');const hash=value=>crypto.createHash('sha256').update(value).digest('hex');const receipt={at:new Date().toISOString(),head:spawnSync('git',['rev-parse','HEAD'],{encoding:'utf8'}).stdout.trim(),platform:process.platform,arch:process.arch,versions:{electron:require('electron/package.json').version,node:process.version,npm:spawnSync('npm',['--version'],{encoding:'utf8'}).stdout.trim()},packagePath:path.dirname(res),sourceStatus:spawnSync('git',['status','--short'],{encoding:'utf8'}).stdout,files:[],producerMatches:[]};
